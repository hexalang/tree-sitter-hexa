}

let x = Union { i: 1 }
let y = Union { s: "hello" }

x.i = 2

================
nested union access
============

@union @struct
class NestedUnion {
	var value Union
	var other Int
}

let nested = NestedUnion { 
	value: Union { i: 123 },
	other: 456
}

nested.value.i = 456
nested.other = 789

================
union with different field types
============

@union
class MixedUnion {
	var number Int
	var text String
	var flag Bool
	var data [UInt8]
}

let mixed = MixedUnion {
	number: 42,
	text: "hello",
	flag: true,
	data: [1, 2, 3]
}

================
union in function parameters
============

fun processUnion(data Union) {
	switch data {
		case Int(value):
			console.log("Processing integer:", value)
		case String(text):
			console.log("Processing string:", text)
		case _:
			console.log("Processing unknown type")
	}
}

================
union assignment patterns
============

let unionValue = Union { i: getIntegerValue() }

switch unionValue {
	case Int(i) if i > 0:
		console.log("Positive integer:", i)
	case Int(i):
		console.log("Non-positive integer:", i)
	case _:
		console.log("Not an integer")
}
