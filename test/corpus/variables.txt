================
readonly keyword
================

readonly var a = 1
readonly let b = "hello"
readonly var c Int = 123
readonly let d String = "world"

// Readonly with arrays
readonly var arr = [1, 2, 3]
readonly let arr2 = ["a", "b", "c"]

// Readonly with objects
readonly var obj = { x: 1, y: 2 }
readonly let obj2 = { name: "test", value: 42 }

// Readonly with nested types
readonly var nested readonly [String] = ["a", "b"]
readonly let complex readonly type { var x Int let y Int } = { x: 1, y: 2 }
readonly let complex readonly interface { var x Int let y Int } = { x: 1, y: 2 }

// Readonly in function parameters
fun process(readonly data [Int]) {
	// data cannot be modified
}

// Readonly return type
fun createReadonly() readonly type { var x Int var y Int } {
	return { x: 1, y: 2 }
}

// Readonly with type inference
readonly var inferred = { x: 1, y: 2 }
