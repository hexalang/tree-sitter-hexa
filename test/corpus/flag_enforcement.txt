============
basic flag enforcement
============

@flagOnce("username_validated")
fun validateUsername(user String) Bool {
	// ... perform checks
	return true
}

@flagRequireOnce("username_validated", "User must be validated before sending welcome email")
fun sendWelcomeEmail(user String) {
	// ... send email
}

@flagRequireOnce("username_validated")
fun logAccess(user String) {
	// ... log access
}

@flagRequireOnce("username_validated")
fun createSession(user String) {
	// ... create session
}

================
flag enforcement with usage
============

fun handleLogin(user String) {
	validateUsername(user) // Emits flag "username_validated"
	sendWelcomeEmail(user) // Requires flag "username_validated"
	logAccess(user) // Requires flag "username_validated"
	createSession(user) // Requires flag "username_validated"
}

================
flag enforcement with error cases
============

// This would cause compiler error
fun brokenLogin(user String) {
	// Missing required flag "username_validated"
	sendWelcomeEmail(user)
	logAccess(user)
	createSession(user)
}

================
nested flag enforcement
============

@flagOnce("outer_locked")
fun lockOuter() {
	// ... actual mutex lock code
}

@flagOnce("inner_locked")
@flagRequireOnce("outer_locked")
fun lockInner() {
	// ... actual inner mutex lock code
}

@flagOnce("inner_unlocked")
@flagRequireOnce("inner_locked")
fun unlockInner() {
	// ... actual inner mutex unlock code
}

@flagDeferRequireOnce("inner_unlocked")
@unflag("inner_locked")
fun unlockOuter() {
	// ... actual outer mutex unlock code
}

================
flag enforcement in complex flow
============

@flagOnce("resource_acquired")
fun processData(data Data) {
	@flagRequireOnce("resource_acquired")
	processData(data)
	@flagDeferRequireOnce("resource_acquired")
	cleanup()
}

@flagOnce("resource_acquired")
fun cleanup() {
	// ... cleanup code
}

================
flag enforcement with custom messages
============

@flagRequireOnce("data_validated", "Input data must pass validation before processing")
fun processData(data Data) {
	validateData(data) // Emits "data_validated"
	transform(data) // Requires "data_validated"
	store(data) // Requires "data_validated"
}

---

================
flag enforcement with conditional flags
============

@flagOnce("initialized")
@flagRequireOnce("initialized", "Component must be initialized before use")
fun useComponent() {
	initializeComponent() // Emits "initialized"
	useComponent() // Requires "initialized"
}

@flagOnce("initialized")
fun resetComponent() {
	@flagDeferRequireOnce("initialized")
	cleanup() // Will be called when "initialized" is still active
	@unflag("initialized")
}

================
flag enforcement with methods
============

class ResourceManager {
	@flagOnce("resource_locked")
	
	fun acquireResource() {
		// ... acquire resource
	}
	
	fun releaseResource() {
		@unflag("resource_locked")
		// ... release resource
	}
}

================
complex flag enforcement scenarios
============

// Multiple flag dependencies
@flagRequireOnce("authenticated", "authorized")
fun accessSecureData() {
	// Requires both "authenticated" and "authorized"
}

// Flag enforcement in loops
@flagOnce("item_processed")
fun processItems(items [Item]) {
	for item in items {
		@flagRequireOnce("item_processed")
			processItem(item)
			@flagDeferRequireOnce("item_processed")
	}
}

		// ... establish connection
	}

		@flagRequireOnce("connected")
		// ... send data
	}
	
	fun disconnect() {
		@unflag("connected")
		// ... close connection
	}
}
