	@verify {
		@save lastProcessed as previousLastProcessed
		lastProcessed >= 0 or "lastProcessed must be non-negative"
	}
	
	let computed = value * 2
	lastProcessed = computed
	return computed
}

fun riskyOperation(value Int) Int {
	@verify {
		@throws {
			// Validate that state did not change on exception
			lastProcessed == previousLastProcessed or "lastProcessed must not change on exception"
		}
	}
	
	if value < 0 {
		throw Error("negative value")
	}
	
	let computed = value * 2
	lastProcessed = computed
	return computed
}

class BankAccount {
	var balance Int = 0
	
	@verify {
		balance >= 0 or "balance cannot be negative"
	}
	
	fun deposit(amount Int) {
		@verify {
			balance + amount >= 0 or "insufficient funds after deposit"
		}
		
		balance += amount
	}
	
	fun withdraw(amount Int) {
		@verify {
			balance - amount >= 0 or "insufficient funds"
		}
		
		balance -= amount
	}
}

class Temperature {
	var celsius Double
	
	@verify {
		celsius >= -273.15 or "temperature above absolute zero"
		celsius <= 1000 or "temperature below maximum"
	}
	
	fun setFahrenheit(fahrenheit Double) {
		@verify {
			(fahrenheit - 32) * 5/9 >= -273.15 or "fahrenheit above absolute zero"
		}
		
		celsius = (fahrenheit - 32) * 5/9
	}
}

fun divide(a Int, b Int) Double? {
	@verify {
		@returns b != 0 or "division by zero not allowed"
		@returns result > 0 or "result must be positive"
	}
	
	if b == 0 {
		return null
	}
	
	return a / b
}
