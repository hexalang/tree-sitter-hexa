status.has(Status.Active)
status.delete(Status.Pending)

================
flag matching in switch
============

switch permissions {
	case Read:
		console.log("Read permission")
	case Write:
		console.log("Write permission")
	case Execute:
		console.log("Execute permission")
	
	// Partial matches
	case Read | Write:
		console.log("Read and Write")
	case Read | _:
		console.log("Has Read permission")
	case _ | not Execute:
		console.log("Does not have Execute permission")
	case Read | Write | _:
		console.log("Has Read or Write permission")
	case Read | Write | Execute:
		console.log("All permissions")
	
	// Dynamic partial match
	case (features) | _:
		console.log("Has specified features")
}

================
complex flag patterns
============

// Exclusion pattern
case Read | _ | not Write:
	console.log("Read but not Write")

// Combination pattern
case Read | Write | Execute:
	console.log("All permissions")

// Dynamic partial match
case (requiredFlags) | _:
	console.log("Has required flags")
