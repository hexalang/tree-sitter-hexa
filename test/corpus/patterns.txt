================
switch ranges
================

switch value {
	case 1:
		console.log("One")
	case x if x > 10:
		console.log("Greater than 10")
	case _ ... 123:
		console.log("Less than or equal to 122")
	case 1 ... 123:
		console.log("Between inclusive 1 and 122")
	case 123 ... _:
		console.log("Greater than or equal to 123")
	case _:
		console.log("Other")
}

================
array patterns
================

switch array {
	case [x, y, z]:
		console.log(x, y, z)
	case [x?, y?, z?]:
		console.log(x, y, z)
	case [x, null, z]:
		console.log(x, z)
	case [1, _, z]:
		console.log(z)
	case [x, ...rest]:
		console.log(x, rest)
	case [x, ..._]:
		console.log(x)
	case _ if array.length == 0:
		console.log("Empty via length check")
	case []:
		console.log("Empty via pattern")
	case [x, [1, _, y]]:
		console.log("Nested pattern", x, y)
	case _:
		console.log("Other")
}

================
map patterns
================

switch map {
	case ["key": "value", "one": "two"]:
		console.log("Match exactly")
	case ["key": "value", ...rest]:
		console.log("Match exactly with rest", rest)
	case ["key": "value", ..._]:
		console.log("Match exactly with rest")
	case [k: "value", ..._]:
		console.log("Match exactly by value", k)
	case [_: v, ..._]:
		console.log("Match at least one key with value", v)
	case [k: v, ..._]:
		console.log("Match at least one key-value pair", k, v)
	case _:
		console.log("Other")
}
