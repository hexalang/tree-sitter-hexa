================
using tuple interfaces
================

let pair Pair<String, Int> = { key: "age", value: 25 }
let triple Triple<String, Int, Bool> = { first: "user", second: 42, third: true }

// Access by index
let name = pair[0]  // "age"
let age = pair[1]    // 25
let user = triple[0] // "user"
let id = triple[1]   // 42
let active = triple[2] // true

// Access by field name
let keyName = pair.key
let keyValue = pair.value
let firstName = triple.first
let secondValue = triple.second
let thirdValue = triple.third
